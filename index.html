Consultas Clave de Detección Forense (SQL / DAX)

-- 1. Detección de Notas Crédito por Devolución SIN Entrada Física en Kardex (Página 2)
SELECT nc.id_nc, nc.fecha_nc, nc.id_cliente, nc.sku, nc.valor_nc, nc.usuario_autoriza
FROM fact_notas_credito nc
INNER JOIN dim_motivo_devolucion m ON nc.id_motivo = m.id_motivo
LEFT JOIN fact_kardex k ON nc.id_nc = k.doc_referencia AND k.tipo_mov = 'ENTRADA_DEVOLUCION'
WHERE m.requiere_ingreso_fisico = TRUE AND k.id_mov IS NULL;
-- 2. Detección de Robo Hormiga: Ajustes Manuales Bajo Umbral de Supervisión ($500,000 COP) (Página 3)
SELECT id_usuario, id_bodega, COUNT(*) AS num_ajustes, AVG(ABS(costo_total)) AS promedio_ajuste, SUM(ABS(costo_total)) AS fuga_total
FROM fact_kardex
WHERE tipo_mov IN ('AJUSTE_AVERIA', 'AJUSTE_INVENTARIO_NEG')
  AND ABS(costo_total) BETWEEN 450000 AND 499999
GROUP BY id_usuario, id_bodega HAVING COUNT(*) >= 5;
-- 3. Medida DAX: Erosión por Ventas Bajo Costo de Reposición Actual (Página 1)
Fuga_Costo_Reposicion =
SUMX(
  fact_ventas,
  VAR CostoUltimaOC = RELATED(dim_productos[costo_ultima_oc])
  VAR CostoContable = fact_ventas[costo_unitario_erp]
  RETURN IF(CostoUltimaOC > CostoContable, (CostoUltimaOC - CostoContable) * fact_ventas[cantidad], 0)
)
